import sqlite3
import os

def ejecutar_transaccion_yos():
    # Ruta basada en tu árbol de directorios
    db_path = os.path.join('Python', 'YosCtr', '_Bdt', 'sistema_yos.db')
    
    # Establecer la conexión
    conn = sqlite3.connect(db_path)
    cursor = conn.cursor()

    try:
        # 1. El inicio de la transacción es automático al ejecutar un comando DML
        cursor.execute("UPDATE cuentas SET saldo = saldo - 100 WHERE id = 1")
        cursor.execute("UPDATE cuentas SET saldo = saldo + 100 WHERE id = 2")

        # 2. Si todo sale bien, confirmamos los cambios
        conn.commit()
        print("Transacción completada con éxito.")

    except sqlite3.Error as e:
        # 3. Si ocurre CUALQUIER error, revertimos todo para evitar datos corruptos
        conn.rollback()
        print(f"Error en la transacción, se realizó Rollback: {e}")

    finally:
        # Cerramos la conexión siempre
        conn.close()

if __name__ == "__main__":
    ejecutar_transaccion_yos()
